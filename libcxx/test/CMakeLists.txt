#
# This file configures the libc++ C++ Standard Library conformance test suite as a
# standalone CMake project. It can be pointed at any prebuilt standard library (or any
# STL implementation) via a Lit site configuration file plus a set of parameters; it does
# not build the library under test itself.
#
# TODO:
# - Handle test/tools (e.g. clang-tidy) so header-lint / module tests get coverage.
#
cmake_minimum_required(VERSION 3.20.0)
project(STLTest CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../runtimes/cmake/Modules")

# Parameters to configure the test suite
set(STLTEST_CONFIG "" CACHE STRING "Path to the (already-resolved) Lit site configuration file to use.")
# The compiler is CMAKE_CXX_COMPILER and the target triple is CMAKE_CXX_COMPILER_TARGET (standard
# CMake toolchain variables), so the tests and this project (GoogleBenchmark) share a toolchain.
# Everything else about the library under test (exceptions, RTTI, hardening, ...) is an opaque set
# of Lit parameters passed via STLTEST_PARAMS.
set(STLTEST_ENABLE_BENCHMARKS "dry-run" CACHE STRING
  "Whether to run the full benchmarks as part of the test suite, to dry-run them or to disable them entirely.
  This requires building GoogleBenchmark. Valid options are 'dry-run', 'run', and 'no'.")
set(STLTEST_PARAMS "" CACHE STRING
  "A list of 'name=value' Lit parameters to bake into the test suite configuration (e.g. 'std=c++26').
  This is a generic passthrough used e.g. to forward LIBCXX_TEST_PARAMS.")

# An optional additional Lit config file to load before the site configuration. This is an opaque
# passthrough: a driver (e.g. the libc++ build) uses it to inject substitutions/parameters specific
# to the library under test, without this project needing to know their content.
set(STLTEST_EXTRA_CONFIG "" CACHE STRING "Optional additional Lit config file to load before the site configuration.")

# The site configuration is the only mandatory input; the compiler comes from CMAKE_CXX_COMPILER
# (which project() guarantees).
if (NOT STLTEST_CONFIG)
  message(FATAL_ERROR "STLTEST_CONFIG must be provided")
endif()

# The target triple: prefer an explicitly-configured CMAKE_CXX_COMPILER_TARGET (e.g. for cross
# compilation); otherwise ask the compiler for its default triple, so GoogleBenchmark and the
# tests target the same thing.
set(_target_triple "${CMAKE_CXX_COMPILER_TARGET}")
if (NOT _target_triple)
  if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine
                    OUTPUT_VARIABLE _target_triple OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  else()
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-target-triple
                    OUTPUT_VARIABLE _target_triple OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
endif()

if (NOT STLTEST_ENABLE_BENCHMARKS STREQUAL "no")
  add_subdirectory(benchmarks)
endif()

# Serialize the Lit parameters into the harness bridge.
include(HandleLitArguments)
set(AUTO_GEN_COMMENT "## Autogenerated by the standalone libc++ test-suite configuration.\n# Do not edit!")
set(SERIALIZED_LIT_PARAMS "# Lit parameters serialized here for llvm-lit to pick them up\n")
set(_params)
# The compiler and target triple come from the CMake toolchain so the tests, GoogleBenchmark, and
# this project all use the same one.
list(APPEND _params "compiler=${CMAKE_CXX_COMPILER}")
list(APPEND _params "target_triple=${_target_triple}")
# The project needs to know whether benchmarks are enabled (it gates building GoogleBenchmark).
list(APPEND _params "enable_benchmarks=${STLTEST_ENABLE_BENCHMARKS}")
# Everything else (std, enable_exceptions, enable_rtti, use_sanitizer, hardening_mode, ...) is an
# opaque set of Lit parameters provided by whoever drives the suite.
list(APPEND _params ${STLTEST_PARAMS})
serialize_lit_params_list(SERIALIZED_LIT_PARAMS _params)

# Generate the harness "bridge" (LitConfig setup, test format, harness substitutions), then the
# top-level lit.site.cfg. It loads the bridge, then an optional driver-provided config fragment
# (STLTEST_EXTRA_CONFIG), then the site configuration describing the library under test.
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/configs/cmake-bridge.cfg.in"
               "${CMAKE_CURRENT_BINARY_DIR}/cmake-bridge.cfg"
               @ONLY)
set(_extra_config_load "")
if (STLTEST_EXTRA_CONFIG)
  set(_extra_config_load "lit_config.load_config(config, r'${STLTEST_EXTRA_CONFIG}')\n")
endif()
string(CONCAT _site_cfg_content
  "# Autogenerated by the standalone libc++ test-suite configuration. Do not edit!\n"
  "lit_config.load_config(config, r'${CMAKE_CURRENT_BINARY_DIR}/cmake-bridge.cfg')\n"
  "${_extra_config_load}"
  "lit_config.load_config(config, r'${STLTEST_CONFIG}')\n"
)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg" CONTENT "${_site_cfg_content}")

#
# Generate a wrapper for Lit that sets up the right config maps.
#
find_package(Python3 REQUIRED COMPONENTS Interpreter)
string(CONCAT _runner_content
  "#!${Python3_EXECUTABLE}\n"
  "import os, platform, sys\n"
  "\n"
  "# Allow generated file to be relocatable.\n"
  "def path(p):\n"
  "    if not p: return ''\n"
  "    # Follows lit.util.abs_path_preserve_drive, which cannot be imported here.\n"
  "    if platform.system() == 'Windows':\n"
  "        return os.path.abspath(os.path.join(os.path.dirname(__file__), p))\n"
  "    else:\n"
  "        return os.path.realpath(os.path.join(os.path.dirname(__file__), p))\n"
  "\n"
  "builtin_parameters = {}\n"
  "builtin_parameters['config_map'] = {\n"
  "  path(r'${CMAKE_CURRENT_SOURCE_DIR}/lit.cfg.py'): path(r'${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg')\n"
  "}\n"
  "\n"
  "# Make sure we can find the lit package.\n"
  "llvm_source_root = path(r'${CMAKE_CURRENT_SOURCE_DIR}/../../llvm')\n"
  "sys.path.insert(0, os.path.join(llvm_source_root, 'utils', 'lit'))\n"
  "if __name__ == '__main__':\n"
  "  from lit.main import main\n"
  "  main(builtin_parameters)\n"
)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stl-runner" CONTENT "${_runner_content}"
     FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)

install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/stl-runner" TYPE BIN)
